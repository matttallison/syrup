cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(syrup_irmidi C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(syrup_irmidi main.c usb_descriptors.c)
pico_generate_pio_header(syrup_irmidi ${CMAKE_CURRENT_LIST_DIR}/irda_tx.pio)
target_include_directories(syrup_irmidi PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(syrup_irmidi PRIVATE pico_stdlib tinyusb_device tinyusb_board hardware_pio hardware_uart)
pico_add_extra_outputs(syrup_irmidi)
